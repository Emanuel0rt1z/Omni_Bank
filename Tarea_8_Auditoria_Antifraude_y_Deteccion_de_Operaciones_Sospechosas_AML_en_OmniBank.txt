de Operadores:

(
    description ILIKE '%Crypto%'

        description IS NULL

    )

2. El Siguiente Nivel Empresarial: 
Purga del String Vacío (NULLIF):
WHERE 
(
    amount >= 10000.00 
    AND tx_type IN ('TRANSFER', 'WITHDRAWAL')
)
OR 
(
    description ILIKE '%Crypto%'
    OR description ILIKE '%offshore%'
    OR description ILIKE '%shell%'
    OR description ILIKE '%casino%'
    OR (
        NULLIF(TRIM(description), '') IS NULL 
        AND tx_type != 'DEPOSIT'
    )
)
-- 3. Presentación Gerencial Limpia (COALESCE + Proyección Quirúrgica):
SELECT
    transaction_id,
    amount,
    tx_type,
    tx_timestamp,
    COALESCE(
        description, 
        'EXCLUSIÓN OPORTUNAMENTE DETECTADA - SIN MOTIVO DE PAGO'
    ) AS descripcion_visible
FROM Core.transactions
WHERE 
(
    amount >= 10000.00 
    AND tx_type IN ('TRANSFER', 'WITHDRAWAL')
)
OR 
(
    description ILIKE '%Crypto%'
    OR description ILIKE '%offshore%'
    OR description ILIKE '%shell%'
    OR description ILIKE '%casino%'
    OR (
        NULLIF(TRIM(description), '') IS NULL 
        AND tx_type != 'DEPOSIT'
    )
)
ORDER BY amount DESC;

-- Breve explicación:
-- Estimado director:
-- El envío masivo de falsos positivos ocurrió porque 
-- en SQL el operador AND se ejecuta antes del OR. 
-- Al no usar paréntesis, la consulta evalúa la condición 
-- description IS NULL por su cuenta y sin 
-- importar el tipo de transacción.
